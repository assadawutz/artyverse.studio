import { createContext, useContext, useEffect, useState, ReactNode } from ‘react’;
export interface CartItem { productId: string; qty: number; }
interface CartCtx { items: CartItem[]; add:(id:string)=>void; remove:(id:string)=>void; setQty:(id:string,qty:number)=>void; clear:()=>void; total:number; }
const Ctx = createContext<CartCtx|null>(null);
const KEY = ‘mb_cart_v1’;
export function CartProvider({ children }:{ children:ReactNode }) {
const [items, setItems] = useState<CartItem[]>([]);
useEffect(() => { try { setItems(JSON.parse(localStorage.getItem(KEY)??”[]”)); } catch { setItems([]); } }, []);
const save = (next:CartItem[]) => { setItems(next); localStorage.setItem(KEY, JSON.stringify(next)); };
const add = (id:string) => { const ex=items.find(i=>i.productId===id); save(ex ? items.map(i=>i.productId===id?{…i,qty:i.qty+1}:i) : […items,{productId:id,qty:1}]); };
const remove = (id:string) => save(items.filter(i=>i.productId!==id));
const setQty = (id:string,qty:number) => save(items.map(i=>i.productId===id?{…i,qty:Math.max(1,qty)}:i));
const clear = () => save([]);
const total = items.reduce((s,i)=>s+i.qty,0);
return <Ctx.Provider value={{items,add,remove,setQty,clear,total}}>{children}</Ctx.Provider>;
}
export const useCart = () => { const c=useContext(Ctx); if(!c) throw new Error(‘useCart outside provider’); return c; };

const Ctx = createContext<ToastCtx|null>(null);

const [msgs, setMsgs] = useState<Msg[]>([]);
